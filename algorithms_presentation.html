private List<Word> selectAdaptiveWords(List<Word> pool, int total) {
    List<Word> hard  = new ArrayList<>();
    List<Word> newW  = new ArrayList<>();
    List<Word> solid = new ArrayList<>();

    for (Word w : pool) {
        if (w.getTotalAttempts() == 0)    newW.add(w);
        else if (w.getMasteryLevel() <= 2) hard.add(w);
        else                               solid.add(w);
    }
    Collections.shuffle(hard);
    Collections.shuffle(newW);
    Collections.shuffle(solid);

    int count     = Math.min(pool.size(), total);
    int wantHard  = (int) Math.round(count * 0.60);
    int wantNew   = (int) Math.round(count * 0.20);
    int wantSolid = count - wantHard - wantNew;

    // Fill + final shuffle to hide group order
    Collections.shuffle(selected);
    return selected;
}

private List<Word> orderByPriority(List<Word> pool) {
    List<Word> hard  = new ArrayList<>(); // Mastery ≤ 2
    List<Word> newW  = new ArrayList<>(); // 0 ניסיונות
    List<Word> solid = new ArrayList<>(); // Mastery 3–5

    for (Word w : pool) {
        if      (w.getTotalAttempts() == 0) newW.add(w);
        else if (w.getMasteryLevel()  <= 2) hard.add(w);
        else                               solid.add(w);
    }
    Collections.shuffle(hard);   // ערבוב בתוך קבוצה
    Collections.shuffle(newW);
    Collections.shuffle(solid);

    // הרכבת הרשימה: קשות → חדשות → חזקות
    List<Word> ordered = new ArrayList<>();
    ordered.addAll(hard);
    ordered.addAll(newW);
    ordered.addAll(solid);
    return ordered;
}

public int[] getWeeklyActivity() {
    int[] result = new int[7];
    Calendar cal = Calendar.getInstance();
    SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd", Locale.US);
    for (int i = 0; i < 7; i++) {
        Calendar day = (Calendar) cal.clone();
        day.add(Calendar.DAY_OF_YEAR, -(6 - i));  // i=0 → לפני 6 ימים
        result[i] = prefs.getInt(KEY_DAILY_CORRECT_PREFIX
                    + sdf.format(day.getTime()), 0);
    }
    return result;
}

חישוב רצף יומי (Streak)

בדיקת daysDiff — אותו יום / יום רצוף / איפוס
📄 StreakManager.java · שורות 89–110

StreakManager.java — computeStreak() + daysDiff() Java 
private static int computeStreak(
        String lastDateStr, int currentStreak, String today) {
    if (lastDateStr == null || lastDateStr.isEmpty())
        return 1;            // פעם ראשונה
    long diffDays = daysDiff(lastDateStr, today);
    if (diffDays == 0) return currentStreak;      // אותו יום
    if (diffDays == 1) return currentStreak + 1;   // רצוף
    return 1;                                      // שבירה — reset
}

private static long daysDiff(String dateA, String dateB) {
    Date a = DATE_FMT.parse(dateA);
    Date b = DATE_FMT.parse(dateB);
    long diffMs = Math.abs(b.getTime() - a.getTime());
    return TimeUnit.MILLISECONDS.toDays(diffMs);
}

    ExifInterface.TAG_ORIENTATION,
    ExifInterface.ORIENTATION_NORMAL);

Matrix matrix = new Matrix();
switch (orientation) {
    case ExifInterface.ORIENTATION_ROTATE_90:  matrix.postRotate(90);  break;
    case ExifInterface.ORIENTATION_ROTATE_180: matrix.postRotate(180); break;
    case ExifInterface.ORIENTATION_ROTATE_270: matrix.postRotate(270); break;
    default: return bitmap; // כבר בכיוון הנכון

    bitmap, 0, 0, bitmap.getWidth(), bitmap.getHeight(),
    matrix, true);
bitmap.recycle(); // פנה זיכרון
return rotated;
