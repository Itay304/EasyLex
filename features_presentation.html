מה מוצג?
כל מילה מוצגת בכרטיס MaterialCardView עם: שם המילה + [חלק הדיבר] אפור קטן יותר, תרגום, משפט דוגמה אנגלי, משפט דוגמה עברי, וכפתור 🔊.

ה-SpannableString מאפשר להציג שני סגנונות שונים בטקסט אחד: המילה בגודל רגיל + חלק הדיבר קטן ואפור, בלי שני TextView נפרדים.

public void bind(final Word word, final OnPronounceClickListener listener) {

    String eng = word.getEnglishWord();
    // חלק הדיבר — אם קיים מוסיפים ב"סוגריים", אחרת מחרוזת ריקה
    String pos = (word.getPartOfSpeech() != null && !word.getPartOfSpeech().isEmpty())
                 ? " (" + word.getPartOfSpeech() + ")" : "";

    // SpannableString = טקסט אחד עם שני סגנונות שונים
    SpannableString spannable = new SpannableString(eng + pos);

    if (!pos.isEmpty()) {
        int start = eng.length();
        // חלק הדיבר: גודל 70% מהגודל הרגיל
        spannable.setSpan(new RelativeSizeSpan(0.7f), start, spannable.length(), 0);
        // חלק הדיבר: צבע אפור
        spannable.setSpan(new ForegroundColorSpan(Color.GRAY), start, spannable.length(), 0);
    }

    tvEnglish.setText(spannable);
    tvHebrew.setText(word.getHebrewTranslation());
    tvExEng.setText(word.getExampleSentence());
    tvExHeb.setText(word.getHebrewExample());

    // כפתור הרמקול — שולח אירוע ל-Fragment ל-TTS
    btnSpeak.setOnClickListener(v -> {
        if (listener != null && word.getEnglishWord() != null)
            listener.onPronounceClick(word.getEnglishWord());
    });
}

private void showResultDialog(int finalScore, int total) {
    if (!isAdded()) return;
    int pct = total > 0 ? finalScore * 100 / total : 0;
    boolean win = pct >= 60; // סף ניצחון: 60%

    View v = getLayoutInflater().inflate(R.layout.dialog_result, null);
    MaterialCardView card = v.findViewById(R.id.resultCard);
    TextView tvEmoji = v.findViewById(R.id.tvResultEmoji);
    TextView tvTitle = v.findViewById(R.id.tvResultTitle);

    // ירוק = ניצחון | אדום = הפסד
    card.setCardBackgroundColor(win ? 0xFF2E7D32 : 0xFFBF360C);
    tvEmoji.setText(win ? "🏆" : "💪");
    tvTitle.setText(win ? "כל הכבוד!" : "אל תוותר!");

    AlertDialog dialog = new AlertDialog.Builder(requireContext())
            .setView(v).setCancelable(false).create();

    dialog.show();

    // אנימציית כניסה: מוקטן ושקוף → מלא עם OvershootInterpolator (קפיצה)
    v.setScaleX(0.75f); v.setScaleY(0.75f); v.setAlpha(0f);
    v.animate().scaleX(1f).scaleY(1f).alpha(1f).setDuration(350)
            .setInterpolator(new OvershootInterpolator()).start();

    // ה-emoji "קופץ" עם השהיה קצרה ו-overshoot חזק יותר
    tvEmoji.setScaleX(0f); tvEmoji.setScaleY(0f);
    tvEmoji.animate().scaleX(1f).scaleY(1f).setDuration(450).setStartDelay(200)
            .setInterpolator(new OvershootInterpolator(3f)).start();
}

פלאשקארדס ממורכזים — cardRestingX

שמירת מיקום טבעי של הכרטיס ב-runtime + חזרה למרכז אחרי גרירה
מה הבעיה?
כשהמשתמש מחליק כרטיסייה ימינה/שמאלה — היא עוזבת את המסך. הכרטיסייה הבאה צריכה להופיע בדיוק במרכז, לא ב-X=0. הבעיה: הפריסה (Layout) ב-XML מוציאה את הכרטיסייה ממוקם על-ידי ConstraintLayout — קואורדינטת X האמיתית נקבעת רק אחרי הצגת המסך (post layout). הפתרון: שמירת cardRestingX פעם אחת אחרי ה-layout הראשון.

/**
 * resetCard — מאפס את הכרטיסייה למצב ברירת מחדל.
 * סדר פעולות קריטי: מסתירים את הצד האחורי קודם,
 * אחר כך מחזירים את הכרטיסייה למיקום — מונע "בזק" של תוכן הפוך.
 */
private void resetCard() {
    flashCard.animate().cancel(); // עצירת כל אנימציה פעילה

    // 1. מסתירים את הצד האחורי קודם — מונע flash
    layoutBack.setRotationY(0);
    layoutBack.setVisibility(View.INVISIBLE);
    layoutFront.setRotationY(0);
    layoutFront.setVisibility(View.VISIBLE);

    // 2. רק עכשיו מחזירים למרכז בטוחה
    flashCard.setX(cardRestingX >= 0 ? cardRestingX : 0); // חזרה למרכז
    flashCard.setRotation(0);  // ישר — ללא הטיה
    flashCard.setAlpha(1);     // גלוי לגמרי
    flashCard.setRotationY(0); // לא מסובב על ציר Y
    isShowingFront = true;
}

/** speakCurrentWord — מקריא את המילה הנוכחית בקול. */
private void speakCurrentWord() {
    // בדיקה שה-TTS לא כובה בהגדרות
    SharedPreferences prefs = requireContext()
            .getSharedPreferences(SettingsFragment.PREFS_SETTINGS, Context.MODE_PRIVATE);
    if (!prefs.getBoolean(SettingsFragment.KEY_TTS_ENABLED, true)) return;

    if (tts != null && currentIndex < quizWords.size()) {
        // QUEUE_FLUSH: מבטל הקראה קודמת מיד ומתחיל חדשה
        tts.speak(quizWords.get(currentIndex).getEnglishWord(),
                 TextToSpeech.QUEUE_FLUSH, null, null);
    }
}

private void calculateMastery(List<Word> words) {
    int total    = words.size();
    int mastered = 0;

    for (Word w : words) {
        // "נכבשה" = לפחות 3 תשובות נכונות
        if (w.getCorrectAttempts() >= 3) mastered++;
    }

    // אחוז שליטה — float לדיוק עשרוני
    float masteryPct = total > 0 ? (mastered * 100.0f) / total : 0f;

    binding.tvMasteryScore.setText(String.format("%.1f%%", masteryPct));
    binding.tvTotalWordsCount.setText(mastered + " מתוך " + total + " מילים נכבשו");

    // עדכון כרטיס הדירוג לפי האחוז
    updateRank(masteryPct, mastered);
}

private void updateRank(float pct, int mastered) {
    String emoji, title, desc;

    if      (pct <  5)  { emoji = "🌱"; title = "מתחיל";  desc = "המסע זה עתה החל!"; }
    else if (pct < 20)  { emoji = "📖"; title = "תלמיד";  desc = "אתה לומד, המשך כך!"; }
    else if (pct < 40)  { emoji = "⭐"; title = "מתקדם";  desc = "ביצועים מרשימים! " + mastered + " מילים נכבשו."; }
    else if (pct < 60)  { emoji = "🎖"; title = "אלוף";   desc = "הכרת מילים רבות! כמעט נסיך!"; }
    else if (pct < 80)  { emoji = "👑"; title = "נסיך";   desc = "בקיאות מרשימה! דרך לכתר!"; }
    else if (pct < 95)  { emoji = "🏆"; title = "מלך";    desc = "שליטה מעולה! כמעט שלמות!"; }

    binding.tvRankEmoji.setText(emoji);
    binding.tvRankTitle.setText(title);
